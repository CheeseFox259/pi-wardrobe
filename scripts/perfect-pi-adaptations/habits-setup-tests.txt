// ── setup.mjs and doctor.mjs integration ──

async function setupFixture(t) {
  const { spawnSync } = await import('node:child_process');
  const { detectPiRuntime } = await import('./scripts/pi-runtime.mjs');
  const home = fixture(t), agentDir = join(home, 'agent');
  const runtime = await detectPiRuntime();
  const env = { PATH: process.env.PATH, HOME: home, PI_CODING_AGENT_DIR: agentDir,
    ...(runtime ? { PI_GLOBAL_NODE_MODULES: runtime.root } : {}) };
  return args => spawnSync(process.execPath, args, { env, encoding: 'utf8', timeout: 30000 });
}

test('setup dry-run succeeds with habits directory', async t => {
  const run = await setupFixture(t);
  const result = run(['setup.mjs', '--dry-run', '--skip-package-install', '--skip-skill-install']);
  assert.equal(result.status, 0, result.stderr);
});

test('doctor reports model habits sync status in an isolated agent directory', async t => {
  const run = await setupFixture(t);
  const synced = run(['setup.mjs', '--skip-package-install', '--skip-skill-install']);
  assert.equal(synced.status, 0, synced.stderr);
  const result = run(['doctor.mjs', '--json']);
  const report = JSON.parse(result.stdout);
  assert.equal(report.statuses['model habits'].status, 'SYNCED');
  assert.notEqual(report.agentDir, process.env.PI_CODING_AGENT_DIR);
});
